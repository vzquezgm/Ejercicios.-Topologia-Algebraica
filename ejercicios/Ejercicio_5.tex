\begin{ejercicio}
    % ID: 5
        Mostrar que un espacio $X$ es contraible si y sólo si existe una homotopía $H:X\times I\to X$ tal que $H(-,0)=\id_X$ y $H(-,1)=c_{x}$, para toda $x\in X$.
    \end{ejercicio}

\begin{demostracion}
$\Rightarrow)$ Supongamos $X\simeq \{*\}$, entonces existen funciones continuas $f:X\to\sing$ 
\end{demostracion}
