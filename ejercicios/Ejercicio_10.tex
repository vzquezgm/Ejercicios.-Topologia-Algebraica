\begin{ejercicio}
    % ID: 10
        Mostrar que $\Vec{\sigma}$ es un homomorfismo de grupo.
    \end{ejercicio}

\begin{demostracion}
Sean $\alpha,\beta\in \Omega(X,x_0)$. Notemos que $\sigma*\overline{\sigma}=c_{x_0}$, entonces 
\begin{align*}
    [\alpha*\beta]&=[\alpha*c_{x_0}*\beta]\\
    &=[\alpha*(\sigma*\overline{\sigma})*\beta].
\end{align*}
Se sigue que,
\begin{align*}
    \Vec{\sigma}\hspace{0.35mm}[\alpha*\beta]&=[\overline{\sigma}*(\alpha*\beta)*\sigma]\\
    &=[\sigma*(\alpha*(\sigma*\overline{\sigma})*\beta)*\sigma]\\
    &=[(\overline{\sigma}*\alpha*\sigma)*(\overline{\sigma}*\beta*\sigma)]\\
    &=[\overline{\sigma}*\alpha*\sigma]*[\overline{\sigma}*\beta*\sigma]\\
    &=\Vec{\sigma}\hspace{0.35mm}[\alpha]*\Vec{\sigma}\hspace{0.35mm}[\beta].
\end{align*}
Por lo tanto, $\Vec{\sigma}$ es un homomorfismo de grupos.
\end{demostracion}
