\begin{ejercicio}
    % ID: 4
        Muestra que si $X\cong Y$ (homeomorfos), entonces $X\simeq Y$. (El regreso no es cierto en general).
    \end{ejercicio}

\begin{demostracion}
    Supongamos $X\cong Y$, entonces existe una función continua biyectiva $f:X\to Y$ tal que $f^{-1}:Y\to X$ es continua. Dado que $f\circ f^{-1}=\id_Y$ y $f^{-1}\circ f=\id_X$, se sigue que $f\circ f^{-1}\simeq\id_Y$ y $f^{-1}\circ f\simeq\id_X$, por lo tanto, $X\simeq Y.$
\end{demostracion}
