\begin{ejercicio}
    % ID: 7
    \label{eje:Pi_1_es_invariante}
        Sean 
        $
            \begin{tikzcd}[column sep=0.5cm]
            (X,x_0) \arrow[r, "f"] & (Y,y_0)\arrow[r, "g"] & (Z,z_0)
        \end{tikzcd}
        $
        funciones continuas basadas. Mostrar que
        \begin{enumerate}
            \item [(1)] ${(g\circ f)}_{\#}=g_{\#}\circ f_{\#}$ y
            \item[(2)] ${\id_{(X,x_0)}}_{\#}=\id_{\Pi_1(X,x_0)}.$ 
        \end{enumerate}
    \end{ejercicio}
\begin{demostracion}
Sea $\alpha:I\to X$. 
\begin{enumerate}[label=(\arabic*)]
    \item Se tiene que
    \begin{align*}
        (g\circ f)_{\#}[\alpha]&=[(f\circ f)\circ \alpha]\\
        &=[g\circ(f\circ\alpha)]\\
        &=g_{\#}[f\circ \alpha]\\
        &=g_{\#}(f_\#[\alpha]).
    \end{align*}
    Por lo tanto, ${(g\circ f)}_{\#}=g_{\#}\circ f_{\#}$.
    \item Por otro lado,
    \begin{align*}
        {\id_{(X,x_0)}}_{\#}[\alpha]&=[\id_{(X,x_0)}\circ\alpha]\\
        &=[\alpha].
    \end{align*}
    Por lo tanto, ${\id_{(X,x_0)}}_{\#}=\id_{\Pi_1(X,x_0)}.$ 
\end{enumerate}
\end{demostracion}
