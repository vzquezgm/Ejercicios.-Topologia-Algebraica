\begin{ejercicio}
    % ID: 10
        Mostrar que $\Vec{\sigma}$ es un homomorfismo de grupo.
    \end{ejercicio}

\begin{demostracion}
Sean $G$ un grupo y $g\in G$. Definamos 
\begin{align*}
    \Vec{g}:G&\to G\\
    x&\mapsto g^{-1}xg.
\end{align*}
 Entonces $\Vec{g}$ es un homomorfismo de grupo. Sea $x,y\in G$, entonces
 \begin{align*}
     \Vec{g}(xy)&=g^{-1}xyg\\
     &=g^{-1}xeyg\\
     &=g^{-1}xgg^{-1}yg\\
     &=\Vec{g}(x)\Vec{g}(y)
 \end{align*}
por lo tanto, $\Vec{g}$ es un homomorfismo de grupo. Notemos que 
$$\Vec{\sigma}[\alpha]=[\overline{\sigma}]*[\alpha]*[{\sigma}], \text{\; para toda } [\alpha]\in\Pi_1(X,x_0).$$ 
Dado que $\Pi_1(X,x_0)$ es un grupo, se sigue que $\Vec{\sigma}$ es un homomorfismo de grupo. 
\end{demostracion}
