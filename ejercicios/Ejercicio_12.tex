\begin{ejercicio}
    % ID: 12
    Mostrar que $\p$ es la inversa de $p^{(X,x_0)
    }$.
\end{ejercicio}

\begin{demostracion}
Sea $\alpha:(I^q,\partial I^q)\to (X,x_0)$ una función continua basada, entonces existe una única función continua $\hat{\alpha}:I^q/\partial I^q\to X$ tal que el siguiente diagrama conmuta
\[
\begin{tikzcd}[column sep=0.65cm, row sep=1cm] 
        I^q\arrow[r,"\alpha"] \arrow[d,"p"'] & X\\
        I^q/\partial I^q \arrow[ur, dashed, "\hat{\alpha}"']
\end{tikzcd}
\]
i.e., $\alpha=\hat{\alpha}\circ p$. Entonces
\begin{align*}
    p
\end{align*}
\end{demostracion}
