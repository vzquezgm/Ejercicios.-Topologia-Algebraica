\begin{ejercicio}
    % ID: 5
        Mostrar que un espacio $X$ es contraible si y sólo si existe una homotopía $H:X\times I\to X$ tal que $H(-,0)=\id_X$ y $H(-,1)=c_{x}$, para toda $x\in X$.
    \end{ejercicio}

\begin{demostracion}
$\Rightarrow)$ Supongamos $X\simeq \{*\}$, entonces existen funciones continuas $f:X\to\sing$ y $g:\sing\to X$ tales que $g\circ f\simeq \id_X$ y $f\circ g\simeq \id_{\sing}.$ Entonces, existen funciones continua $H_1:X\times I\to X$ y $H_2:\sing\times I\to\sing$ tales que 
\begin{align*}
    H_1(x,0)&=g(f(x))=g(*),\\
    H_1(x,1)&=\id_X,\\
    H_2(*,0)&=f(g(*)),\\
    H_2(*,1)&=\id_{\sing}(*)=*.
\end{align*}
   Definamos $H:X\times I\to X$
\end{demostracion}
